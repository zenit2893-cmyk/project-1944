cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(cod3_timing_observer LANGUAGES CXX)
    find_package(rexglue 0.10.0.5 REQUIRED CONFIG)
    set(COD3_TIMING_STANDALONE ON)
endif()

if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "Timing observer currently requires Windows and the verified Clang toolchain")
endif()
if(NOT TARGET rex::runtime)
    message(FATAL_ERROR "Find the pinned ReXGlue SDK before adding integration/timing")
endif()

add_library(cod3_timing_observer STATIC candidate_observer.cpp)
add_library(cod3::timing_observer ALIAS cod3_timing_observer)
target_compile_features(cod3_timing_observer PUBLIC cxx_std_23)
target_include_directories(cod3_timing_observer PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(cod3_timing_observer PUBLIC rex::runtime)
target_compile_options(cod3_timing_observer PRIVATE -ffp-model=strict -fno-strict-aliasing -fwrapv)

add_library(cod3_timing_hooks OBJECT override_hooks.cpp)
target_link_libraries(cod3_timing_hooks PUBLIC cod3::timing_observer)
target_compile_options(cod3_timing_hooks PRIVATE -ffp-model=strict)

function(cod3_enable_timing_hooks target_name)
    if(NOT TARGET "${target_name}")
        message(FATAL_ERROR "Unknown timing hook host target ${target_name}")
    endif()
    target_sources("${target_name}" PRIVATE $<TARGET_OBJECTS:cod3_timing_hooks>)
    target_link_libraries("${target_name}" PRIVATE cod3::timing_observer)
endfunction()

option(COD3_TIMING_BUILD_TESTS "Build synthetic native passthrough tests, not gameplay" ${COD3_TIMING_STANDALONE})
if(COD3_TIMING_BUILD_TESTS)
    enable_testing()
    add_executable(cod3_timing_passthrough_test tests/passthrough_test.cpp)
    cod3_enable_timing_hooks(cod3_timing_passthrough_test)
    target_compile_options(cod3_timing_passthrough_test PRIVATE -ffp-model=strict)
    file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." _timing_workspace)
    add_test(NAME timing_native_passthrough
        COMMAND cod3_timing_passthrough_test "${_timing_workspace}" "${CMAKE_CURRENT_BINARY_DIR}/test-artifacts")
endif()
