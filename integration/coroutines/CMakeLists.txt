cmake_minimum_required(VERSION 3.25)
project(cod3_coroutines LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET rex::runtime)
  find_package(rexglue CONFIG REQUIRED)
endif()
add_library(cod3_coroutines SHARED cod3_coroutines.cpp)
target_compile_definitions(cod3_coroutines PRIVATE COD3_COROUTINES_BUILD)
target_include_directories(cod3_coroutines PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(cod3_coroutines PUBLIC rex::runtime)
target_compile_options(cod3_coroutines PRIVATE -ffp-model=strict -fno-strict-aliasing -fwrapv)

function(cod3_attach_capture_hooks target_name source_file)
  if(NOT TARGET ${target_name})
    message(FATAL_ERROR "Missing CoD3 mission target ${target_name}")
  endif()
  target_sources(${target_name} PRIVATE ${source_file})
  target_link_libraries(${target_name} PRIVATE cod3_coroutines)
endfunction()

function(cod3_enable_coroutines host_target entry_output_directory)
  target_sources(${host_target} PRIVATE "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/main_hooks.cpp")
  target_include_directories(${host_target} PRIVATE ${entry_output_directory})
  target_link_libraries(${host_target} PRIVATE cod3_coroutines)
  if(NOT TARGET ${host_target}_recomp)
    message(FATAL_ERROR "Missing main recompiled object target ${host_target}_recomp")
  endif()
  # The custom PCH is compiled by the object target, not just the final EXE.
  target_link_libraries(${host_target}_recomp PRIVATE cod3_coroutines)
  rexglue_configure_module_target(cod3_coroutines HOST ${host_target})
  include("${CMAKE_CURRENT_FUNCTION_LIST_DIR}/generated/capture_targets.cmake")
endfunction()
